cmake_minimum_required(VERSION 3.10)
project(nu_native_runtime C)
set(CMAKE_C_STANDARD 90)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(nu_runtime STATIC nu_runtime.c nu_cache.c nu_translator.c nu_package.c)
target_include_directories(nu_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(nu_runtime_tests tests/test_runtime.c)
target_link_libraries(nu_runtime_tests nu_runtime)
add_executable(nu_pack nu_pack.c)
target_link_libraries(nu_pack nu_runtime)
enable_testing()
add_test(NAME nu_runtime_tests COMMAND nu_runtime_tests)
if(WIN32)
 add_executable(nu_launcher WIN32 win32_host.c)
 target_link_libraries(nu_launcher nu_runtime comdlg32)
endif()
